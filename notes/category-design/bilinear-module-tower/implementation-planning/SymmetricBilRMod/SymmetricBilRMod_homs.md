# Homs: Symmetric Bilinear Module Morphisms and Orthogonal Groups

Morphism implementation for symmetric bilinear modules focusing on orthogonal transformations and isometries.

## Homset Structure

```python
from sage.categories.homsets import HomsetWithBase

class SymmetricBilinearModuleHomset(BilinearModuleHomset):
    """
    Homset of morphisms between symmetric bilinear modules.
    
    A morphism f: (M₁, b₁) → (M₂, b₂) between symmetric bilinear modules
    is an R-module homomorphism that preserves the symmetric bilinear form:
    b₂(f(v), f(w)) = b₁(v, w) for all v, w ∈ M₁.
    
    For symmetric forms, this gives rise to:
    - Orthogonal transformations (isometries of the same space)
    - Proper orthogonal transformations (determinant +1)
    - Reflections (determinant -1)
    - Similarity transformations (scaling by constants)
    
    The collection of isometries forms the orthogonal group O(M, b).
    
    EXAMPLES::
    
        sage: G1 = matrix(QQ, [[1, 0], [0, 1]])  # Standard inner product
        sage: G2 = matrix(QQ, [[2, 0], [0, 2]])  # Scaled inner product
        sage: M1 = SymmetricBilinearModule(G1)
        sage: M2 = SymmetricBilinearModule(G2)
        sage: H = Hom(M1, M2)
        sage: H
        Set of Morphisms from Symmetric bilinear module of rank 2 over Rational Field
                             to Symmetric bilinear module of rank 2 over Rational Field
                             in Category of symmetric bilinear modules over Rational Field
    """
    
    def __init__(self, domain, codomain, category=None, base=None, check=True):
        """Initialize homset of symmetric bilinear module morphisms."""
        if category is None:
            from sage.categories.symmetric_bilinear_modules import SymmetricBilinearModules
            category = SymmetricBilinearModules(domain.base_ring())
        
        super().__init__(domain, codomain, category=category, base=base, check=check)
    
    def _repr_(self):
        """String representation of the homset."""
        return (f"Set of Morphisms from {self.domain()} to {self.codomain()} "
                f"in Category of symmetric bilinear modules over {self.base()}")
    
    def orthogonal_group(self):
        """
        Return the orthogonal group of endomorphisms (when domain = codomain).
        
        O(M, b) = {f ∈ GL(M) : b(f(v), f(w)) = b(v, w) for all v, w}
        
        OUTPUT:
        OrthogonalGroup object
        
        EXAMPLES::
        
            sage: G = matrix(QQ, [[1, 0], [0, 1]])
            sage: M = SymmetricBilinearModule(G)
            sage: H = Hom(M, M)
            sage: O = H.orthogonal_group()
            sage: O
            Orthogonal Group of degree 2 over Rational Field
        """
        if self.domain() != self.codomain():
            raise ValueError("Orthogonal group only defined for endomorphisms")
        
        from sage.groups.matrix_gps.orthogonal import OrthogonalMatrixGroup
        domain = self.domain()
        return OrthogonalMatrixGroup(
            domain.rank(),
            domain.base_ring(),
            invariant_bilinear_form=domain.gram_matrix()
        )
    
    def special_orthogonal_group(self):
        """
        Return the special orthogonal group SO(M, b).
        
        SO(M, b) = {f ∈ O(M, b) : det(f) = 1}
        
        OUTPUT:
        SpecialOrthogonalGroup object
        """
        if self.domain() != self.codomain():
            raise ValueError("Special orthogonal group only defined for endomorphisms")
        
        from sage.groups.matrix_gps.orthogonal import SO
        domain = self.domain()
        return SO(domain.rank(), domain.base_ring(), invariant_form=domain.gram_matrix())
```

## Orthogonal Transformation Implementation

```python
from sage.categories.morphism import Morphism

class OrthogonalTransformation(BilinearModuleMorphism):
    """
    Orthogonal transformation of a symmetric bilinear module.
    
    An orthogonal transformation is an isometry f: M → M that preserves
    the symmetric bilinear form: b(f(v), f(w)) = b(v, w).
    
    For symmetric bilinear forms, these satisfy:
    - f^T G f = G where G is the Gram matrix
    - det(f) = ±1
    - f is invertible with f^(-1) also orthogonal
    
    Classification:
    - det(f) = +1: Proper orthogonal (rotation)
    - det(f) = -1: Improper orthogonal (reflection/rotation-reflection)
    
    EXAMPLES::
    
        sage: # 2D rotation matrix
        sage: from sage.symbolic.ring import var
        sage: theta = var('theta')
        sage: R = matrix(SR, [[cos(theta), -sin(theta)], [sin(theta), cos(theta)]])
        sage: G = matrix(QQ, [[1, 0], [0, 1]])
        sage: M = SymmetricBilinearModule(G)
        sage: # rot = OrthogonalTransformation(Hom(M, M), matrix=R)
    """
    
    def __init__(self, homset, f=None, matrix=None):
        """Initialize orthogonal transformation with automatic verification."""
        super().__init__(homset, f=f, matrix=matrix, check=True)
        
        if not self.is_orthogonal_transformation():
            raise ValueError("Morphism is not an orthogonal transformation")
    
    def is_orthogonal_transformation(self):
        """Verify that this transformation preserves the symmetric form."""
        if self.domain() != self.codomain():
            return False
        
        # Check f^T G f = G
        A = self.matrix()
        G = self.domain().gram_matrix()
        
        return A.transpose() * G * A == G
    
    def determinant(self):
        """
        Return determinant (±1 for orthogonal transformations).
        
        EXAMPLES::
        
            sage: G = matrix(QQ, [[1, 0], [0, 1]])
            sage: M = SymmetricBilinearModule(G)
            sage: # Rotation by 90 degrees
            sage: R = matrix(QQ, [[0, -1], [1, 0]])
            sage: rot = OrthogonalTransformation(Hom(M, M), matrix=R)
            sage: rot.determinant()
            1
        """
        return self.matrix().determinant()
    
    def is_proper_orthogonal(self):
        """Test if determinant is +1 (proper rotation)."""
        return self.determinant() == 1
    
    def is_improper_orthogonal(self):
        """Test if determinant is -1 (reflection or rotation-reflection)."""
        return self.determinant() == -1
    
    def inverse(self):
        """
        Return inverse orthogonal transformation.
        
        For orthogonal transformations, A^(-1) = A^T.
        
        OUTPUT:
        OrthogonalTransformation representing the inverse
        """
        if not hasattr(self, '_inverse'):
            # For orthogonal matrices: A^(-1) = A^T
            inv_matrix = self.matrix().transpose()
            self._inverse = OrthogonalTransformation(
                self.parent().reversed(), matrix=inv_matrix
            )
        return self._inverse
    
    def conjugate_by(self, other):
        """
        Conjugate this transformation by another orthogonal transformation.
        
        Returns other * self * other^(-1).
        
        INPUT:
        - other -- another orthogonal transformation
        
        OUTPUT:
        OrthogonalTransformation representing the conjugate
        """
        if not isinstance(other, OrthogonalTransformation):
            raise ValueError("Can only conjugate by orthogonal transformations")
        
        # Compute other * self * other^(-1)
        conjugate_matrix = other.matrix() * self.matrix() * other.inverse().matrix()
        return OrthogonalTransformation(self.parent(), matrix=conjugate_matrix)
    
    def axis_angle_decomposition(self):
        """
        Decompose rotation into axis and angle (for 3D proper orthogonal).
        
        Only applies to 3×3 proper orthogonal transformations.
        
        OUTPUT:
        Dictionary with axis vector and rotation angle
        """
        if self.domain().rank() != 3:
            raise ValueError("Axis-angle decomposition only for 3D transformations")
        
        if not self.is_proper_orthogonal():
            raise ValueError("Axis-angle decomposition only for proper rotations")
        
        A = self.matrix()
        
        # Find eigenvalues and eigenvectors
        eigenvals, eigenvecs = A.eigenmatrix_right()
        
        # Look for eigenvalue 1 (rotation axis)
        for i, ev in enumerate(eigenvals.diagonal()):
            if abs(ev - 1) < 1e-10:  # Eigenvalue 1
                axis = eigenvecs.column(i)
                break
        else:
            raise ValueError("No rotation axis found")
        
        # Compute rotation angle using trace formula
        # tr(R) = 1 + 2*cos(θ) for 3D rotation
        trace = A.trace()
        cos_angle = (trace - 1) / 2
        
        from sage.functions.trig import arccos
        angle = arccos(cos_angle)
        
        return {
            'axis': self.domain()._from_vector(axis),
            'angle': angle,
            'axis_vector': axis
        }


class Reflection(OrthogonalTransformation):
    """
    Reflection across a hyperplane in a symmetric bilinear module.
    
    A reflection across hyperplane H_n = {v : b(v,n) = 0} is given by:
    ref_n(v) = v - 2 * b(v,n)/b(n,n) * n
    
    Properties:
    - Determinant = -1 (improper orthogonal)
    - Involution: ref_n² = identity  
    - Fixes the hyperplane pointwise
    - Negates the normal vector: ref_n(n) = -n
    
    EXAMPLES::
    
        sage: G = matrix(QQ, [[1, 0], [0, 1]])
        sage: M = SymmetricBilinearModule(G)
        sage: e, f = M.gens()
        sage: # Reflection across line perpendicular to e (x-axis reflection)
        sage: refl = Reflection(Hom(M, M), normal=e)
        sage: refl.determinant()
        -1
        sage: refl(f) == f  # f is in hyperplane
        True
        sage: refl(e) == -e  # e is normal vector
        True
    """
    
    def __init__(self, homset, normal=None, matrix=None):
        """
        Initialize reflection.
        
        INPUT:
        - homset -- homset of endomorphisms
        - normal -- normal vector to hyperplane (non-isotropic)
        - matrix -- explicit reflection matrix (alternative)
        """
        if matrix is not None:
            super().__init__(homset, matrix=matrix)
        elif normal is not None:
            # Construct reflection matrix from normal vector
            if normal.is_isotropic():
                raise ValueError("Cannot reflect across hyperplane with isotropic normal")
            
            self._normal = normal
            matrix = self._construct_reflection_matrix(normal)
            super().__init__(homset, matrix=matrix)
        else:
            raise ValueError("Must provide either normal vector or matrix")
    
    def _construct_reflection_matrix(self, normal):
        """
        Construct reflection matrix from normal vector.
        
        Formula: R = I - 2 * (n ⊗ n) / b(n,n)
        where ⊗ is outer product.
        """
        domain = self.domain()
        n_coords = normal.to_vector()
        G = domain.gram_matrix()
        
        # Compute b(n,n)
        n_norm_sq = n_coords * G * n_coords
        
        if n_norm_sq == 0:
            raise ValueError("Normal vector is isotropic")
        
        # Construct reflection matrix: I - 2 * G * (n ⊗ n) / b(n,n)
        from sage.matrix.constructor import identity_matrix
        I = identity_matrix(domain.base_ring(), domain.rank())
        
        # Outer product G * n * n^T
        outer_product = G * n_coords.column() * n_coords.row()
        
        reflection_matrix = I - 2 * outer_product / n_norm_sq
        
        return reflection_matrix
    
    def normal_vector(self):
        """Return the normal vector to the reflection hyperplane."""
        if hasattr(self, '_normal'):
            return self._normal
        else:
            # Extract normal from matrix (find eigenvector with eigenvalue -1)
            A = self.matrix()
            eigenvals, eigenvecs = A.eigenmatrix_right()
            
            for i, ev in enumerate(eigenvals.diagonal()):
                if abs(ev + 1) < 1e-10:  # Eigenvalue -1
                    normal_coords = eigenvecs.column(i)
                    return self.domain()._from_vector(normal_coords)
            
            raise ValueError("Could not determine normal vector")
    
    def hyperplane(self):
        """
        Return description of the reflection hyperplane.
        
        OUTPUT:
        Dictionary describing the hyperplane
        """
        normal = self.normal_vector()
        
        return {
            'normal': normal,
            'equation': f"b(v, {normal}) = 0",
            'dimension': self.domain().rank() - 1,
            'codimension': 1
        }
    
    def is_involution(self):
        """Verify that this reflection is an involution (R² = I)."""
        R = self.matrix()
        I = R.parent().identity_matrix()
        return R * R == I
    
    def conjugate_reflection(self, other_normal):
        """
        Conjugate this reflection by reflection in another hyperplane.
        
        The composition of two reflections is either:
        - Identity (if hyperplanes are the same)
        - Rotation (if hyperplanes intersect at non-zero angle)
        
        INPUT:
        - other_normal -- normal vector to other hyperplane
        
        OUTPUT:
        OrthogonalTransformation representing the composition
        """
        other_refl = Reflection(self.parent(), normal=other_normal)
        composition = other_refl * self * other_refl
        return composition


class Rotation(OrthogonalTransformation):
    """
    Rotation in a symmetric bilinear module.
    
    A rotation is a proper orthogonal transformation (determinant +1)
    that preserves orientation. In 2D, parameterized by angle.
    In 3D, parameterized by axis and angle.
    
    Properties:
    - Determinant = +1
    - Preserves all distances and angles
    - Can be written as product of even number of reflections
    - Forms connected component of orthogonal group
    """
    
    def __init__(self, homset, angle=None, axis=None, matrix=None):
        """
        Initialize rotation.
        
        INPUT:
        - homset -- homset of endomorphisms  
        - angle -- rotation angle (for 2D or with axis)
        - axis -- rotation axis (for 3D)
        - matrix -- explicit rotation matrix
        """
        if matrix is not None:
            super().__init__(homset, matrix=matrix)
        elif angle is not None:
            if homset.domain().rank() == 2:
                matrix = self._construct_2d_rotation(angle)
            elif axis is not None and homset.domain().rank() == 3:
                matrix = self._construct_3d_rotation(axis, angle)
            else:
                raise ValueError("Need axis for 3D rotation or dimension 2 for angle-only")
            
            super().__init__(homset, matrix=matrix)
        else:
            raise ValueError("Must provide angle (and axis for 3D) or matrix")
        
        if not self.is_proper_orthogonal():
            raise ValueError("Rotation must have determinant +1")
    
    def _construct_2d_rotation(self, angle):
        """Construct 2D rotation matrix."""
        from sage.functions.trig import cos, sin
        
        return matrix(self.domain().base_ring(), [
            [cos(angle), -sin(angle)],
            [sin(angle),  cos(angle)]
        ])
    
    def _construct_3d_rotation(self, axis, angle):
        """Construct 3D rotation matrix using Rodrigues' formula."""
        # Rodrigues' rotation formula: R = I + sin(θ)[v]× + (1-cos(θ))[v]×²
        # where [v]× is the skew-symmetric cross-product matrix
        
        axis_coords = axis.to_vector()
        axis_normalized = axis_coords / axis_coords.norm()
        
        from sage.functions.trig import cos, sin
        from sage.matrix.constructor import identity_matrix, matrix
        
        I = identity_matrix(self.domain().base_ring(), 3)
        
        # Cross-product matrix [v]×
        v = axis_normalized
        cross_matrix = matrix(self.domain().base_ring(), [
            [0, -v[2], v[1]],
            [v[2], 0, -v[0]],
            [-v[1], v[0], 0]
        ])
        
        # Rodrigues' formula
        R = I + sin(angle) * cross_matrix + (1 - cos(angle)) * cross_matrix * cross_matrix
        
        return R
    
    def rotation_angle(self):
        """Return the rotation angle."""
        if self.domain().rank() == 2:
            # For 2D: angle = arctan2(R[1,0], R[0,0])
            R = self.matrix()
            from sage.functions.trig import atan2
            return atan2(R[1,0], R[0,0])
        else:
            # For 3D: use axis-angle decomposition
            decomp = self.axis_angle_decomposition()
            return decomp['angle']
    
    def rotation_axis(self):
        """Return the rotation axis (for 3D rotations)."""
        if self.domain().rank() != 3:
            raise ValueError("Rotation axis only defined for 3D")
        
        decomp = self.axis_angle_decomposition()
        return decomp['axis']
```

## Similarity Transformations

```python
class SimilarityTransformation(Morphism):
    """
    Similarity transformation between symmetric bilinear modules.
    
    A similarity is a morphism f: (M₁, b₁) → (M₂, b₂) such that:
    b₂(f(v), f(w)) = λ * b₁(v, w) for some scalar λ > 0.
    
    This includes:
    - Isometries (λ = 1)
    - Homotheties (scaling by λ)
    - Compositions of isometries and scalings
    
    Similarities preserve angles but not necessarily distances.
    """
    
    def __init__(self, homset, f=None, matrix=None, scaling_factor=None):
        """
        Initialize similarity transformation.
        
        INPUT:
        - homset -- homset between symmetric bilinear modules
        - f -- function defining transformation
        - matrix -- matrix representation  
        - scaling_factor -- the similarity ratio λ
        """
        super().__init__(homset)
        
        if matrix is not None:
            self._matrix = matrix
        elif f is not None:
            self._function = f
        else:
            raise ValueError("Must provide function or matrix")
        
        if scaling_factor is not None:
            self._scaling_factor = scaling_factor
        else:
            # Compute scaling factor from matrix
            self._scaling_factor = self._compute_scaling_factor()
    
    def _compute_scaling_factor(self):
        """Compute the scaling factor λ from the transformation matrix."""
        # For similarity: f^T G₂ f = λ G₁
        # So λ = (f^T G₂ f)[0,0] / G₁[0,0] (using first basis element)
        
        domain = self.domain()
        codomain = self.codomain()
        
        A = self.matrix()
        G1 = domain.gram_matrix()
        G2 = codomain.gram_matrix()
        
        # Check similarity condition: A^T G₂ A = λ G₁
        product = A.transpose() * G2 * A
        
        # Find λ by comparing with G₁
        if G1[0,0] != 0:
            lambda_val = product[0,0] / G1[0,0]
        else:
            # Find first non-zero diagonal entry
            for i in range(G1.nrows()):
                if G1[i,i] != 0:
                    lambda_val = product[i,i] / G1[i,i]
                    break
            else:
                raise ValueError("Cannot determine scaling factor for degenerate form")
        
        # Verify similarity condition
        expected = lambda_val * G1
        if not (product - expected).is_zero():
            raise ValueError("Transformation is not a similarity")
        
        return lambda_val
    
    def scaling_factor(self):
        """Return the similarity scaling factor λ."""
        return self._scaling_factor
    
    def is_isometry(self):
        """Test if this is an isometry (λ = 1)."""
        return self._scaling_factor == 1
    
    def is_homothety(self):
        """Test if this is a pure scaling (f(v) = √λ * v)."""
        # Check if matrix is scalar multiple of identity
        A = self.matrix()
        sqrt_lambda = self._scaling_factor.sqrt()
        I = A.parent().identity_matrix()
        
        return A == sqrt_lambda * I
    
    def orthogonal_part(self):
        """
        Extract orthogonal part of similarity transformation.
        
        Every similarity can be written as f = √λ * O where O is orthogonal.
        
        OUTPUT:
        OrthogonalTransformation representing the orthogonal part
        """
        sqrt_lambda = self._scaling_factor.sqrt()
        orthogonal_matrix = self.matrix() / sqrt_lambda
        
        # Create homset for orthogonal transformation
        # Need to scale the target form by λ to make it an isometry
        return OrthogonalTransformation(
            self.parent(), 
            matrix=orthogonal_matrix
        )
```

```python
def stabilizer_subgroup(element):
    """
    Return stabilizer subgroup of an element.
    
    Stab(v) = {g ∈ O(M) : g(v) = v}
    
    For non-zero v, this is isomorphic to O(M/⟨v⟩).
    
    INPUT:
    - element -- element to stabilize
    
    OUTPUT:
    Subgroup of orthogonal group
    """
    if element.is_zero():
        # Stabilizer of zero is the whole orthogonal group
        return element.parent().orthogonal_group()
    
    # For non-zero v, stabilizer is O(v⊥)
    perp_space = element.parent().orthogonal_complement(element.parent().span([element]))
    return perp_space.orthogonal_group()

def orbit_space(element):
    """
    Describe the orbit of an element under orthogonal group action.
    
    For element v, orbit is {g·v : g ∈ O(M)}.
    
    Classification:
    - If q(v) > 0: orbit is sphere of radius √q(v)
    - If q(v) < 0: orbit is "pseudo-sphere" 
    - If q(v) = 0, v ≠ 0: orbit is light cone minus origin
    - If v = 0: orbit is {0}
    
    INPUT:
    - element -- element whose orbit to describe
    
    OUTPUT:
    Dictionary describing the orbit structure
    """
    if element.is_zero():
        return {'type': 'point', 'description': 'singleton {0}'}
    
    q_val = element.quadratic_form()
    
    if q_val > 0:
        return {
            'type': 'sphere',
            'radius': q_val.sqrt(),
            'dimension': element.parent().rank() - 1,
            'description': f'Sphere of radius {q_val.sqrt()}'
        }
    elif q_val < 0:
        return {
            'type': 'pseudo_sphere', 
            'radius': (-q_val).sqrt(),
            'dimension': element.parent().rank() - 1,
            'description': f'Pseudo-sphere of radius {(-q_val).sqrt()}'
        }
    else:  # q_val == 0, element != 0
        return {
            'type': 'light_cone_sheet',
            'dimension': element.parent().rank() - 1,
            'description': 'Light cone minus origin'
        }
```

## Mathematical Properties

The symmetric bilinear module morphism framework maintains these properties:

```python
# Mathematical assertion: Orthogonal transformation characterization
# f is orthogonal ⟺ f^T G f = G ⟺ b(f(v), f(w)) = b(v, w)

# Mathematical assertion: Determinant property
# det(f) = ±1 for orthogonal transformations

# Mathematical assertion: Group structure
# O(M, b) forms a group under composition

# Mathematical assertion: Cartan-Dieudonné theorem
# Every orthogonal transformation is product of ≤ n reflections

# Mathematical assertion: Spectral decomposition
# Symmetric matrices diagonalize with orthogonal eigenvector matrices

# Mathematical assertion: Similarity invariants
# Similarities preserve angles but scale distances by √λ

# Mathematical assertion: Orbit-stabilizer theorem
```
