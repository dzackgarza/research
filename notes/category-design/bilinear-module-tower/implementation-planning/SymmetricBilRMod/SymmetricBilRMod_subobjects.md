# Subobjects: Symmetric Bilinear Module Submodules and Lattice Operations

Submodule implementation for symmetric bilinear modules with emphasis on lattice theory, orthogonal complements, and isotropic subspaces.

## Submodule Structure

```python
from sage.modules.submodule import Submodule_free_ambient

class SymmetricBilinearSubmodule(BilinearSubmodule):
    """
    Submodule of a symmetric bilinear module with inherited symmetric structure.
    
    A submodule W ⊆ M inherits the restriction of the symmetric bilinear form:
    b_W(v,w) = b_M(v,w) for v,w ∈ W
    
    This gives W its own symmetric bilinear structure, plus relationships
    to the ambient module through orthogonal complements and projections.
    
    Key submodule types in symmetric bilinear modules:
    - Orthogonal complements: W^⊥ = {v ∈ M : b(v,w) = 0 for all w ∈ W}
    - Isotropic submodules: b(v,w) = 0 for all v,w ∈ W
    - Anisotropic submodules: no non-zero isotropic vectors
    - Definite submodules: positive or negative definite restriction
    
    EXAMPLES::
    
        sage: G = matrix(QQ, [[2, 1, 0], [1, 3, 1], [0, 1, 2]])
        sage: M = SymmetricBilinearModule(G)
        sage: e, f, g = M.gens()
        sage: W = M.submodule([e, f])
        sage: W.ambient_module() is M
        True
        sage: W.is_symmetric()
        True
        sage: W.dimension()
        2
        
        sage: # Orthogonal complement
        sage: W_perp = W.orthogonal_complement()
        sage: W_perp.dimension()
        1
        sage: all(W.bilinear_form(v, w) == 0 for v in W.gens() for w in W_perp.gens())
        True
    """
    
    def __init__(self, ambient, gens, category=None, **kwds):
        """Initialize symmetric bilinear submodule."""
        if category is None:
            from sage.categories.symmetric_bilinear_modules import SymmetricBilinearModules
            category = SymmetricBilinearModules(ambient.base_ring()).Subobjects()
        
        super().__init__(ambient, gens, category=category, **kwds)
    
    def _repr_(self):
        """String representation."""
        return (f"Submodule of {self.ambient_module()} "
                f"generated by {len(self.gens())} elements")
    
    def is_symmetric(self):
        """Symmetric bilinear submodules inherit symmetry."""
        return True
    
    def gram_matrix(self):
        """
        Gram matrix of this submodule.
        
        Computes b(g_i, g_j) where {g_i} are the generators.
        
        OUTPUT:
        Symmetric matrix of bilinear form values on generators
        
        EXAMPLES::
        
            sage: G = matrix(QQ, [[2, 1], [1, 3]])
            sage: M = SymmetricBilinearModule(G)
            sage: e, f = M.gens()
            sage: W = M.submodule([e + f, 2*e - f])
            sage: W.gram_matrix()
            [10 -1]
            [-1  8]
        """
        ambient = self.ambient_module()
        gens = self.gens()
        n = len(gens)
        
        from sage.matrix.constructor import matrix
        gram = matrix(self.base_ring(), n, n)
        
        for i in range(n):
            for j in range(i, n):
                value = ambient.bilinear_form(gens[i], gens[j])
                gram[i,j] = value
                gram[j,i] = value  # Symmetry
        
        return gram
    
    def restricted_form(self):
        """
        The bilinear form restricted to this submodule.
        
        Returns a new SymmetricBilinearModule representing the restriction.
        
        OUTPUT:
        SymmetricBilinearModule with Gram matrix of restriction
        """
        from sage.modules.symmetric_bilinear_module import SymmetricBilinearModule
        return SymmetricBilinearModule(self.gram_matrix())
    
    def quadratic_form(self, v):
        """Quadratic form evaluation on submodule element."""
        return self.ambient_module().quadratic_form(v)
    
    def signature(self):
        """Signature of the restricted form."""
        return self.restricted_form().signature()
    
    def is_positive_definite(self):
        """Test if restriction is positive definite."""
        return self.signature()[1] == 0 and self.signature()[2] == 0
    
    def is_negative_definite(self):
        """Test if restriction is negative definite."""
        return self.signature()[0] == 0 and self.signature()[2] == 0
    
    def is_definite(self):
        """Test if restriction is definite."""
        return self.is_positive_definite() or self.is_negative_definite()
    
    def is_indefinite(self):
        """Test if restriction is indefinite."""
        p, q, r = self.signature()
        return p > 0 and q > 0
    
    def is_degenerate(self):
        """Test if restriction is degenerate."""
        return self.signature()[2] > 0
    
    def witt_index(self):
        """Witt index of the restricted form."""
        p, q, r = self.signature()
        return min(p, q)
```

## Orthogonal Complements

```python
def orthogonal_complement(self):
    """
    Return the orthogonal complement W^⊥.
    
    W^⊥ = {v ∈ M : b(v,w) = 0 for all w ∈ W}
    
    For non-degenerate forms: dim(W) + dim(W^⊥) = dim(M)
    
    OUTPUT:
    SymmetricBilinearSubmodule representing W^⊥
    
    EXAMPLES::
    
        sage: G = matrix(QQ, [[1, 0, 0], [0, 1, 0], [0, 0, -1]])
        sage: M = SymmetricBilinearModule(G)
        sage: e, f, g = M.gens()
        sage: W = M.submodule([e, f])  # Positive definite plane
        sage: W_perp = W.orthogonal_complement()
        sage: W_perp.gens()
        [g]  # Negative definite line
        sage: W.dimension() + W_perp.dimension() == M.dimension()
        True
    """
    ambient = self.ambient_module()
    W_gens = self.gens()
    
    # Find vectors orthogonal to all generators of W
    # This involves solving the system b(v, w_i) = 0 for all w_i ∈ W
    
    ambient_gens = ambient.gens()
    n = len(ambient_gens)
    m = len(W_gens)
    
    from sage.matrix.constructor import matrix
    
    # Build constraint matrix: each row is coefficients of b(x, w_i) = 0
    # where x = Σ c_j * ambient_gen_j
    constraints = matrix(ambient.base_ring(), m, n)
    
    for i, w in enumerate(W_gens):
        for j, v in enumerate(ambient_gens):
            constraints[i, j] = ambient.bilinear_form(v, w)
    
    # Find null space of constraint matrix
    null_space = constraints.right_kernel()
    
    # Convert null space basis to ambient module elements
    orthogonal_gens = []
    for null_vec in null_space.basis():
        element = sum(c * v for c, v in zip(null_vec, ambient_gens) if c != 0)
        if not element.is_zero():
            orthogonal_gens.append(element)
    
    return ambient.submodule(orthogonal_gens)

def orthogonal_sum(self, other):
    """
    Orthogonal direct sum with another submodule.
    
    Returns W ⊕ V where W ⊥ V (bilinear form vanishes between them).
    
    INPUT:
    - other -- another submodule orthogonal to this one
    
    OUTPUT:
    SymmetricBilinearSubmodule representing W ⊕ V
    
    EXAMPLES::
    
        sage: G = matrix(QQ, [[1, 0, 0], [0, 1, 0], [0, 0, -1]])
        sage: M = SymmetricBilinearModule(G)
        sage: e, f, g = M.gens()
        sage: W1 = M.submodule([e])
        sage: W2 = M.submodule([f])
        sage: W_sum = W1.orthogonal_sum(W2)
        sage: W_sum.dimension()
        2
        sage: span([e, f]) == W_sum  # Check spanning sets match
        True
    """
    # Verify orthogonality
    for v in self.gens():
        for w in other.gens():
            if self.ambient_module().bilinear_form(v, w) != 0:
                raise ValueError("Submodules are not orthogonal")
    
    # Combine generators
    combined_gens = list(self.gens()) + list(other.gens())
    return self.ambient_module().submodule(combined_gens)

def is_orthogonal_to(self, other):
    """
    Test if this submodule is orthogonal to another.
    
    Returns True if b(v,w) = 0 for all v ∈ self, w ∈ other.
    
    INPUT:
    - other -- another submodule or element
    
    OUTPUT:
    Boolean
    """
    if hasattr(other, 'gens'):
        # other is a submodule
        return all(self.ambient_module().bilinear_form(v, w) == 0 
                  for v in self.gens() for w in other.gens())
    else:
        # other is an element
        return all(self.ambient_module().bilinear_form(v, other) == 0 
                  for v in self.gens())

def orthogonal_projection_onto(self, v):
    """
    Orthogonal projection of element v onto this submodule.
    
    proj_W(v) = Σᵢ (b(v,wᵢ)/b(wᵢ,wᵢ)) * wᵢ where {wᵢ} is orthogonal basis.
    
    INPUT:
    - v -- element of ambient module
    
    OUTPUT:
    Element of this submodule representing the projection
    
    EXAMPLES::
    
        sage: G = matrix(QQ, [[1, 0, 0], [0, 1, 0], [0, 0, 1]])
        sage: M = SymmetricBilinearModule(G)
        sage: e, f, g = M.gens()
        sage: W = M.submodule([e, f])  # xy-plane
        sage: v = 3*e + 4*f + 5*g
        sage: proj = W.orthogonal_projection_onto(v)
        sage: proj
        3*e + 4*f
        sage: (v - proj).is_orthogonal_to(W)
        True
    """
    # Get orthogonal basis for this submodule
    orthogonal_basis = self.orthogonal_basis()
    
    projection = self.ambient_module().zero()
    
    for w in orthogonal_basis:
        w_norm_sq = self.ambient_module().quadratic_form(w)
        if w_norm_sq != 0:
            coeff = self.ambient_module().bilinear_form(v, w) / w_norm_sq
            projection += coeff * w
    
    return projection

def orthogonal_basis(self):
    """
    Return orthogonal basis for this submodule.
    
    Uses Gram-Schmidt process with the ambient bilinear form.
    
    OUTPUT:
    List of mutually orthogonal elements spanning this submodule
    """
    gens = self.gens()
    orthogonal = []
    ambient = self.ambient_module()
    
    for v in gens:
        # Subtract projections onto previous orthogonal vectors
        orthogonal_v = v
        for u in orthogonal:
            u_norm_sq = ambient.quadratic_form(u)
            if u_norm_sq != 0:
                proj_coeff = ambient.bilinear_form(v, u) / u_norm_sq
                orthogonal_v = orthogonal_v - proj_coeff * u
        
        if not orthogonal_v.is_zero():
            orthogonal.append(orthogonal_v)
    
    return orthogonal
```

## Isotropic and Anisotropic Submodules

```python
def is_isotropic(self):
    """
    Test if this is an isotropic submodule.
    
    A submodule W is isotropic if b(v,w) = 0 for all v,w ∈ W.
    Equivalently, all vectors in W are isotropic.
    
    OUTPUT:
    Boolean
    
    EXAMPLES::
    
        sage: # Hyperbolic plane
        sage: H = matrix(QQ, [[0, 1], [1, 0]])
        sage: M = SymmetricBilinearModule(H)
        sage: e, f = M.gens()
        sage: # Both e and f are isotropic
        sage: W1 = M.submodule([e])
        sage: W1.is_isotropic()
        True
        sage: # But span{e,f} is not totally isotropic
        sage: W2 = M.submodule([e, f])
        sage: W2.is_isotropic()
        False  # b(e,f) = 1 ≠ 0
    """
    gram = self.gram_matrix()
    return gram.is_zero()

def is_anisotropic(self):
    """
    Test if this is an anisotropic submodule.
    
    A submodule is anisotropic if it has no non-zero isotropic vectors.
    Equivalently, the restricted form is anisotropic.
    
    OUTPUT:
    Boolean
    """
    return self.restricted_form().is_anisotropic()

def maximal_isotropic_extension(self):
    """
    Find maximal isotropic submodule containing this one.
    
    If this submodule is isotropic, extends it to a maximal isotropic
    submodule. If not isotropic, raises an error.
    
    OUTPUT:
    SymmetricBilinearSubmodule that is maximal isotropic
    """
    if not self.is_isotropic():
        raise ValueError("Can only extend isotropic submodules")
    
    # Algorithm: repeatedly add vectors from orthogonal complement
    # that maintain isotropy property
    current = self
    ambient = self.ambient_module()
    
    while True:
        # Find orthogonal complement
        perp = current.orthogonal_complement()
        
        # Look for vectors in perp that maintain isotropy
        found_extension = False
        
        for v in perp.gens():
            # Check if adding v maintains isotropy
            if ambient.quadratic_form(v) == 0:
                # v is isotropic, can add it
                extended_gens = list(current.gens()) + [v]
                extended = ambient.submodule(extended_gens)
                
                if extended.is_isotropic():
                    current = extended
                    found_extension = True
                    break
        
        if not found_extension:
            break
    
    return current

def anisotropic_part(self):
    """
    Extract anisotropic part of this submodule.
    
    Decomposes W = W_iso ⊕ W_anis where W_iso is maximal isotropic
    subspace and W_anis is anisotropic.
    
    OUTPUT:
    Tuple (isotropic_part, anisotropic_part)
    """
    # This requires sophisticated algorithm from quadratic form theory
    # Simplified version: find radical and factor it out
    
    restricted = self.restricted_form()
    p, q, r = restricted.signature()
    
    if r == 0:
        # Non-degenerate: find Witt decomposition
        w = min(p, q)  # Witt index
        
        if w == 0:
            # Already anisotropic
            return (self.ambient_module().submodule([]), self)
        else:
            # Has isotropic part - need algorithm to extract it
            raise NotImplementedError("Witt decomposition algorithm")
    else:
        # Degenerate: radical is automatically isotropic
        radical = self.radical()
        # Quotient by radical gives anisotropic part
        raise NotImplementedError("Radical quotient construction")

def radical(self):
    """
    Return the radical (degenerate part) of this submodule.
    
    Rad(W) = {v ∈ W : b(v,w) = 0 for all w ∈ W}
    
    OUTPUT:
    SymmetricBilinearSubmodule representing the radical
    """
    # Radical consists of vectors orthogonal to entire submodule
    return self.intersection(self.orthogonal_complement())
```

```python
def lattice_span(self):
    """
    Z-span of this submodule (for integral forms).
    
    Returns the Z-lattice generated by the current generators,
    preserving the integral quadratic form structure.
    
    OUTPUT:
    IntegralLattice representing the Z-span
    """
    if not self.ambient_module().is_integral():
        raise ValueError("Lattice span only defined for integral forms")
    
    # This would construct an IntegralLattice object
    from sage.modules.integral_lattice import IntegralLattice
    return IntegralLattice(self.gram_matrix())

def hermite_normal_form(self):
    """
    Hermite normal form of lattice generators.
    
    For integral lattices, computes HNF of generator matrix
    to get canonical basis.
    
    OUTPUT:
    Tuple (hnf_basis, transformation_matrix)
    """
    if not self.ambient_module().is_integral():
        raise ValueError("HNF only defined for integral lattices")
    
    # Get generator coordinates as matrix
    gen_matrix = matrix([g.to_vector() for g in self.gens()])
    
    # Compute Hermite normal form
    hnf, U = gen_matrix.hermite_form(transformation=True)
    
    # Convert back to module elements
    ambient = self.ambient_module()
    hnf_gens = [ambient._from_vector(row) for row in hnf.rows() if not row.is_zero()]
    
    return hnf_gens, U
```

```python
def orthogonal_direct_sum_decomposition(self):
    """
    Decompose into orthogonal direct sum.
    
    W = W₁ ⊕ W₂ ⊕ ... ⊕ Wₖ where Wᵢ ⊥ Wⱼ for i ≠ j
    and each Wᵢ is indecomposable.
    
    OUTPUT:
    List of indecomposable orthogonal submodules
    """
    # Use spectral decomposition of restricted Gram matrix
    restricted = self.restricted_form()
    basis, eigenvals = restricted.diagonalize()
    
    # Group by eigenvalue to form orthogonal components
    eigenvalue_groups = {}
    for v, ev in zip(basis, eigenvals):
        if ev not in eigenvalue_groups:
            eigenvalue_groups[ev] = []
        eigenvalue_groups[ev].append(v)
    
    # Each eigenspace is an orthogonal component
    components = []
    ambient = self.ambient_module()
    
    for eigenval, eigenvecs in eigenvalue_groups.items():
        if eigenval != 0:  # Skip zero eigenvalue (radical)
            component = ambient.submodule(eigenvecs)
            components.append(component)
    
    return components

def witt_decomposition(self):
    """
    Witt decomposition of this submodule.
    
    W = H^⊕w ⊕ A ⊕ R where:
    - H is hyperbolic plane
    - w is Witt index  
    - A is anisotropic part
    - R is radical
    
    OUTPUT:
    Dictionary with components of Witt decomposition
    """
    restricted = self.restricted_form()
    return restricted.witt_decomposition()

def intersection(self, other):
    """
    Intersection with another submodule.
    
    Returns W ∩ V as a symmetric bilinear submodule.
    
    INPUT:
    - other -- another submodule of same ambient space
    
    OUTPUT:
    SymmetricBilinearSubmodule representing intersection
    """
    # Find basis for intersection of vector spaces
    self_space = self.vector_space()
    other_space = other.vector_space()
    intersection_space = self_space.intersection(other_space)
    
    # Convert intersection basis back to ambient module elements
    ambient = self.ambient_module()
    intersection_gens = []
    
    for basis_vector in intersection_space.basis():
        element = ambient._from_vector(basis_vector)
        intersection_gens.append(element)
    
    return ambient.submodule(intersection_gens)

def sum(self, other):
    """
    Sum with another submodule: W + V = span(W ∪ V).
    
    INPUT:
    - other -- another submodule
    
    OUTPUT:
    SymmetricBilinearSubmodule representing the sum
    """
    combined_gens = list(self.gens()) + list(other.gens())
    return self.ambient_module().submodule(combined_gens)
```

## Mathematical Properties

The symmetric bilinear submodule framework maintains these properties:

```python
# Mathematical assertion: Orthogonal complement dimension
# dim(W) + dim(W^⊥) = dim(M) for non-degenerate forms

# Mathematical assertion: Double orthogonal complement
# W^⊥⊥ = W for non-degenerate forms

# Mathematical assertion: Isotropic subspace bound  
# dim(isotropic subspace) ≤ Witt index of ambient form

# Mathematical assertion: Witt decomposition
# Every submodule decomposes as hyperbolic ⊕ anisotropic ⊕ radical

# Mathematical assertion: Radical properties
# Rad(W) = W ∩ W^⊥ and Rad(W) consists of degenerate vectors

# Mathematical assertion: Lattice properties
# For integral forms, Z-span preserves integrality and signature

# Mathematical assertion: Orthogonal sum
# (W ⊕ V)^⊥ = W^⊥ ∩ V^⊥ when W ⊥ V

# Mathematical assertion: Signature additivity
# signature(W ⊕ V) = signature(W) + signature(V) when W ⊥ V
```

This submodule framework provides comprehensive lattice-theoretic operations while maintaining the geometric structure of symmetric bilinear forms and enabling connections to quadratic form theory, Witt groups, and arithmetic geometry.
