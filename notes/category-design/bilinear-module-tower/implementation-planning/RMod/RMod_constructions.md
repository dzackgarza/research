# Constructions: R-Module Factory Functions and Standard Constructions

User-friendly factory functions for constructing R-modules with various properties and standard constructions.

## Main Factory Function

```python
def RModule(base_ring, *args, **kwds):
    """
    Main factory function for constructing R-modules.
    
    This provides multiple construction patterns:
    1. RModule(R) - Returns category RModules(R)
    2. RModule(R, n) - Creates free module of rank n
    3. RModule(R, basis=...) - Creates module with specified basis
    4. RModule(R, gens=...) - Creates module with generators
    5. RModule(R, relations=...) - Creates quotient by relations
    
    INPUT:
    - base_ring -- ring R for the R-module structure
    - *args -- positional arguments determining construction type
    - **kwds -- keyword arguments for specific constructions
    
    OUTPUT:
    R-module or category of R-modules
    
    EXAMPLES::
    
        sage: # Category of R-modules
        sage: RModule(ZZ)
        Category of modules over Integer Ring
        
        sage: # Free module of rank 3
        sage: M = RModule(ZZ, 3)
        sage: M
        Free module of rank 3 over Integer Ring
        sage: M.gens()
        (e0, e1, e2)
        
        sage: # Module with named basis
        sage: M = RModule(QQ, basis=['x', 'y', 'z'])
        sage: M
        Free module with basis ['x', 'y', 'z'] over Rational Field
        
        sage: # Module with generators (possibly dependent)
        sage: M = RModule(ZZ, gens=[[1, 2], [3, 4], [5, 6]])
        sage: M
        Submodule of rank 2 of Free module of rank 2 over Integer Ring
        
        sage: # Quotient by relations
        sage: F = RModule(ZZ, 2)
        sage: e0, e1 = F.gens()
        sage: M = RModule(ZZ, relations=[2*e0, 3*e1])
        sage: M
        Quotient of Free module of rank 2 over Integer Ring by relations
    """
    if not args and not kwds:
        # Return category
        from sage.categories.modules import Modules
        return Modules(base_ring)
    
    # Dispatch based on arguments
    if len(args) == 1 and isinstance(args[0], (int, Integer)):
        # RModule(R, n) - free module of rank n
        return FreeModule(base_ring, args[0], **kwds)
    
    elif 'basis' in kwds:
        # RModule(R, basis=...) - module with specified basis
        return FreeModule_with_basis(base_ring, basis=kwds['basis'], **kwds)
    
    elif 'gens' in kwds:
        # RModule(R, gens=...) - submodule generated by elements
        gens = kwds.pop('gens')
        ambient_dim = max(len(g) if hasattr(g, '__len__') else 1 for g in gens)
        ambient = FreeModule(base_ring, ambient_dim)
        return ambient.submodule(gens, **kwds)
    
    elif 'relations' in kwds:
        # RModule(R, relations=...) - quotient by relations
        relations = kwds.pop('relations')
        if len(args) > 0:
            # Free module specified
            free_module = args[0]
        else:
            # Infer from relations
            free_module = relations[0].parent()
        
        submodule = free_module.submodule(relations)
        return free_module.quotient(submodule, **kwds)
    
    else:
        # Default to free module construction
        return FreeModule(base_ring, *args, **kwds)


class FreeModule(Parent):
    """
    Free R-module of finite rank.
    
    Standard representation as R^n with canonical basis e_0, ..., e_{n-1}.
    
    EXAMPLES::
    
        sage: M = FreeModule(ZZ, 3)
        sage: M
        Free module of rank 3 over Integer Ring
        sage: M.is_free()
        True
        sage: M.rank()
        3
        sage: M.gens()
        (e0, e1, e2)
    """
    
    def __init__(self, base_ring, rank, **kwds):
        """Initialize free module R^n."""
        self._base_ring = base_ring
        self._rank = rank
        
        from sage.categories.modules import Modules
        category = Modules(base_ring).Free().FinitelyGenerated()
        
        Parent.__init__(self, base=base_ring, category=category, **kwds)
    
    def _repr_(self):
        """String representation."""
        return f"Free module of rank {self._rank} over {self._base_ring}"
    
    def rank(self):
        """Return rank of free module."""
        return self._rank
    
    def gens(self):
        """Return canonical generators."""
        return tuple(self.gen(i) for i in range(self._rank))
    
    def gen(self, i):
        """Return i-th canonical generator."""
        if not 0 <= i < self._rank:
            raise IndexError(f"Generator index {i} out of range [0, {self._rank})")
        
        coords = [self._base_ring.zero()] * self._rank
        coords[i] = self._base_ring.one()
        return self.element_class(self, coords)
    
    def basis(self):
        """Return canonical basis (same as generators for free modules)."""
        return self.gens()
    
    def _element_constructor_(self, x):
        """Construct element from various formats."""
        if isinstance(x, (list, tuple)) and len(x) == self._rank:
            # Direct coordinate list
            return self.element_class(self, x)
        elif hasattr(x, 'parent') and x.parent() is self:
            # Already an element
            return x
        else:
            # Try to convert
            return self.element_class(self, x)


class FreeModule_with_basis(FreeModule):
    """
    Free R-module with specified basis indexing.
    
    Instead of e_0, e_1, ..., uses custom basis labels.
    
    EXAMPLES::
    
        sage: M = FreeModule_with_basis(QQ, basis=['x', 'y', 'z'])
        sage: M
        Free module with basis ['x', 'y', 'z'] over Rational Field
        sage: x, y, z = M.basis()
        sage: 2*x + 3*y - z
        2*x + 3*y - z
    """
    
    def __init__(self, base_ring, basis=None, rank=None, **kwds):
        """Initialize with custom basis."""
        if basis is not None:
            self._basis_keys = basis
            rank = len(basis)
        elif rank is not None:
            self._basis_keys = list(range(rank))
        else:
            raise ValueError("Must specify either basis or rank")
        
        super().__init__(base_ring, rank, **kwds)
        
        # Update category to WithBasis
        from sage.categories.modules import Modules
        self._refine_category_(Modules(base_ring).WithBasis())
    
    def _repr_(self):
        """String representation."""
        return f"Free module with basis {self._basis_keys} over {self._base_ring}"
    
    def basis_keys(self):
        """Return basis indexing set."""
        return self._basis_keys
    
    def basis(self):
        """Return basis as family indexed by keys."""
        from sage.sets.family import Family
        return Family(self._basis_keys, self.basis_element)
    
    def basis_element(self, key):
        """Return basis element for given key."""
        if key not in self._basis_keys:
            raise KeyError(f"Key {key} not in basis")
        
        index = self._basis_keys.index(key)
        return self.gen(index)
```

## Standard Constructions

```python
def zero_module(base_ring):
    """
    Construct the zero module {0}.
    
    The unique module with one element.
    
    INPUT:
    - base_ring -- base ring R
    
    OUTPUT:
    Zero R-module
    
    EXAMPLES::
    
        sage: Z = zero_module(ZZ)
        sage: Z
        Zero module over Integer Ring
        sage: Z.is_zero()
        True
        sage: Z.rank()
        0
    """
    return FreeModule(base_ring, 0)


def cyclic_module(base_ring, generator_annihilator=None):
    """
    Construct cyclic module R/I where I = Ann(generator).
    
    A cyclic module is generated by a single element.
    
    INPUT:
    - base_ring -- base ring R
    - generator_annihilator -- ideal I such that module is R/I
    
    OUTPUT:
    Cyclic R-module
    
    EXAMPLES::
    
        sage: # Z/6Z as Z-module
        sage: M = cyclic_module(ZZ, 6)
        sage: M
        Cyclic module Z/6Z
        sage: M.order()
        6
        
        sage: # R as cyclic R-module (I = 0)
        sage: R = cyclic_module(ZZ)
        sage: R.is_free()
        True
    """
    if generator_annihilator is None:
        # R itself as cyclic module
        return FreeModule(base_ring, 1)
    
    # Construct as quotient R/I
    R = FreeModule(base_ring, 1)
    I = base_ring.ideal(generator_annihilator)
    
    # Relations are I * generator
    relations = [i * R.gen(0) for i in I.gens()]
    return R.quotient(R.submodule(relations))


def direct_sum(*modules):
    """
    Direct sum M₁ ⊕ M₂ ⊕ ... ⊕ Mₙ of modules.
    
    Elements are tuples (m₁, m₂, ..., mₙ) with mᵢ ∈ Mᵢ.
    
    INPUT:
    - *modules -- R-modules over same base ring
    
    OUTPUT:
    Direct sum as R-module
    
    EXAMPLES::
    
        sage: M1 = RModule(ZZ, 2)
        sage: M2 = RModule(ZZ, 3)
        sage: M = direct_sum(M1, M2)
        sage: M
        Direct sum of Free module of rank 2 over Integer Ring
                   and Free module of rank 3 over Integer Ring
        sage: M.rank()
        5
        
        sage: # Elements as tuples
        sage: e1 = M1.gen(0)
        sage: f2 = M2.gen(1)
        sage: M((e1, f2))
        (e0, 0) ⊕ (0, e1, 0)
    """
    if not modules:
        raise ValueError("Need at least one module")
    
    base_ring = modules[0].base_ring()
    if not all(M.base_ring() == base_ring for M in modules):
        raise ValueError("All modules must have same base ring")
    
    from sage.modules.free_module import FreeModule_direct_sum
    return FreeModule_direct_sum(modules)


def tensor_product(*modules):
    """
    Tensor product M₁ ⊗ M₂ ⊗ ... ⊗ Mₙ over base ring.
    
    Universal bilinear/multilinear construction.
    
    INPUT:
    - *modules -- R-modules over same base ring
    
    OUTPUT:
    Tensor product as R-module
    
    EXAMPLES::
    
        sage: M = RModule(ZZ, 2)
        sage: N = RModule(ZZ, 3)
        sage: T = tensor_product(M, N)
        sage: T
        Tensor product of Free module of rank 2 over Integer Ring
                      and Free module of rank 3 over Integer Ring
        sage: T.rank()
        6  # 2 × 3
        
        sage: # Tensor of basis elements
        sage: e0, e1 = M.gens()
        sage: f0, f1, f2 = N.gens()
        sage: e0.tensor(f1) in T
        True
    """
    if not modules:
        raise ValueError("Need at least one module")
    
    if len(modules) == 1:
        return modules[0]
    
    # Use Sage's tensor product construction
    from sage.modules.tensor_operations import TensorProductOfModules
    return TensorProductOfModules(modules)


def exterior_power(module, degree):
    """
    Exterior power ∧^k(M) of a module.
    
    Antisymmetric k-tensors modulo alternating relations.
    
    INPUT:
    - module -- R-module
    - degree -- non-negative integer k
    
    OUTPUT:
    k-th exterior power as R-module
    
    EXAMPLES::
    
        sage: M = RModule(QQ, 3)
        sage: E2 = exterior_power(M, 2)
        sage: E2
        2nd exterior power of Free module of rank 3 over Rational Field
        sage: E2.rank()
        3  # (3 choose 2)
        
        sage: # Basis consists of e_i ∧ e_j for i < j
        sage: e0, e1, e2 = M.gens()
        sage: E2.basis()
        (e0∧e1, e0∧e2, e1∧e2)
    """
    if degree < 0:
        raise ValueError("Degree must be non-negative")
    
    if degree == 0:
        # ∧^0(M) = R
        return FreeModule(module.base_ring(), 1)
    
    if degree == 1:
        # ∧^1(M) = M
        return module
    
    if module.is_free() and module.rank() < degree:
        # ∧^k(M) = 0 if k > rank
        return zero_module(module.base_ring())
    
    # Use Sage's exterior algebra construction
    from sage.algebras.exterior_algebra import ExteriorAlgebra
    E = ExteriorAlgebra(module)
    return E.degree_module(degree)


def symmetric_power(module, degree):
    """
    Symmetric power Sym^k(M) of a module.
    
    Symmetric k-tensors modulo commutation relations.
    
    INPUT:
    - module -- R-module
    - degree -- non-negative integer k
    
    OUTPUT:
    k-th symmetric power as R-module
    
    EXAMPLES::
    
        sage: M = RModule(QQ, 2)
        sage: S3 = symmetric_power(M, 3)
        sage: S3
        3rd symmetric power of Free module of rank 2 over Rational Field
        sage: S3.rank()
        4  # (2+3-1 choose 3) = (4 choose 3)
        
        sage: # Basis consists of monomials of degree 3
        sage: e0, e1 = M.gens()
        sage: S3.basis()
        (e0^3, e0^2*e1, e0*e1^2, e1^3)
    """
    if degree < 0:
        raise ValueError("Degree must be non-negative")
    
    if degree == 0:
        # Sym^0(M) = R
        return FreeModule(module.base_ring(), 1)
    
    if degree == 1:
        # Sym^1(M) = M
        return module
    
    # Use polynomial ring construction
    from sage.rings.polynomial.polynomial_ring_constructor import PolynomialRing
    
    if module.is_free():
        # For free modules, Sym(M) ≅ R[x₁, ..., xₙ]
        n = module.rank()
        P = PolynomialRing(module.base_ring(), n, 'x')
        # Return degree-k homogeneous component
        return P.degree_module(degree)
    else:
        raise NotImplementedError("Symmetric power for non-free modules")


def dual_module(module):
    """
    Dual module M* = Hom_R(M, R).
    
    Linear functionals on M with R-module structure.
    
    INPUT:
    - module -- R-module
    
    OUTPUT:
    Dual module M*
    
    EXAMPLES::
    
        sage: M = RModule(ZZ, 3)
        sage: D = dual_module(M)
        sage: D
        Dual of Free module of rank 3 over Integer Ring
        sage: D.rank()
        3
        
        sage: # Dual basis
        sage: e0, e1, e2 = M.gens()
        sage: f0, f1, f2 = D.basis()
        sage: f0(e0)
        1
        sage: f0(e1)
        0
    """
    from sage.modules.free_module_homspace import FreeModuleDual
    
    if module.is_free() and hasattr(module, 'rank'):
        return FreeModuleDual(module)
    else:
        # General construction as Hom(M, R)
        R = FreeModule(module.base_ring(), 1)
        return Hom(module, R)
```

```python
def module_from_matrix_space(matrix_space):
    """
    View matrix space as R-module.
    
    M_{m×n}(R) is naturally an R-module of dimension mn.
    
    INPUT:
    - matrix_space -- space of matrices over R
    
    OUTPUT:
    R-module structure on matrix space
    
    EXAMPLES::
    
        sage: MS = MatrixSpace(QQ, 2, 3)
        sage: M = module_from_matrix_space(MS)
        sage: M
        Free module of rank 6 over Rational Field from 2×3 matrices
        sage: M.rank()
        6
        
        sage: # Conversion between representations
        sage: A = matrix(QQ, [[1, 2, 3], [4, 5, 6]])
        sage: v = M(A)
        sage: v.to_vector()
        (1, 2, 3, 4, 5, 6)
    """
    m, n = matrix_space.dims()
    base_ring = matrix_space.base_ring()
    rank = m * n
    
    # Create wrapper module
    module = FreeModule(base_ring, rank)
    
    # Add conversion methods
    def matrix_to_vector(mat):
        return vector(base_ring, mat.list())
    
    def vector_to_matrix(vec):
        return matrix_space(vec.list())
    
    module._matrix_space = matrix_space
    module.matrix_to_vector = matrix_to_vector
    module.vector_to_matrix = vector_to_matrix
    
    return module


def module_from_ideal(ideal):
    """
    View ideal as R-module.
    
    Every ideal I ⊆ R is naturally an R-module.
    
    INPUT:
    - ideal -- ideal in ring R
    
    OUTPUT:
    R-module structure on ideal
    
    EXAMPLES::
    
        sage: R = ZZ
        sage: I = R.ideal([6, 10, 15])
        sage: M = module_from_ideal(I)
        sage: M
        Ideal (1) of Integer Ring as R-module
        sage: M.gens()
        (1,)  # Since gcd(6,10,15) = 1
    """
    base_ring = ideal.ring()
    
    # Ideals are cyclic modules for PIDs
    if base_ring.is_principal_ideal_domain():
        gen = ideal.gen()  # Single generator
        if gen == 0:
            return zero_module(base_ring)
        elif gen == 1:
            return FreeModule(base_ring, 1)
        else:
            # Construct as submodule of R
            R = FreeModule(base_ring, 1)
            return R.submodule([gen * R.gen(0)])
    else:
        # General ideal - use generators
        R = FreeModule(base_ring, 1)
        submodule_gens = [g * R.gen(0) for g in ideal.gens()]
        return R.submodule(submodule_gens)


def module_from_ring(ring, base_ring=None):
    """
    View ring as module over subring.
    
    Every ring is a module over itself or any subring.
    
    INPUT:
    - ring -- ring to view as module
    - base_ring -- subring to use as base (default: ring itself)
    
    OUTPUT:
    Module structure on ring
    
    EXAMPLES::
    
        sage: # Gaussian integers as Z-module
        sage: R = ZZ[I]
        sage: M = module_from_ring(R, base_ring=ZZ)
        sage: M
        Gaussian Integer Ring as module over Integer Ring
        sage: M.rank()
        2  # Basis {1, I}
        
        sage: # Polynomial ring as module
        sage: P = PolynomialRing(QQ, 'x')
        sage: M = module_from_ring(P, base_ring=QQ)
        sage: M.basis()
        (1, x, x^2, x^3, ...)  # Infinite basis
    """
    if base_ring is None:
        base_ring = ring
    
    if not base_ring.is_subring(ring):
        raise ValueError(f"{base_ring} must be a subring of {ring}")
    
    # Determine if finitely generated
    if hasattr(ring, 'gens') and ring.gens():
        # Ring with finite generators
        ring_gens = ring.gens()
        ambient_rank = len(ring_gens)
        
        # Create ambient free module
        ambient = FreeModule(base_ring, ambient_rank)
        
        # Ring elements as module generators
        return ambient.span(ring_gens)
    else:
        # Infinite or special case
        raise NotImplementedError(f"Module structure for {ring} over {base_ring}")
```

```python
def quotient_by_relations(generators, relations, base_ring=None):
    """
    Construct module by generators and relations.
    
    Creates F/R where F is free on generators and R is
    submodule generated by relations.
    
    INPUT:
    - generators -- list of generator names or number
    - relations -- list of linear combinations to quotient by
    - base_ring -- base ring (inferred if not given)
    
    OUTPUT:
    Finitely presented module
    
    EXAMPLES::
    
        sage: # Z-module with generators x,y and relation 2x + 3y = 0
        sage: M = quotient_by_relations(['x', 'y'], ['2*x + 3*y'], base_ring=ZZ)
        sage: M
        Finitely presented module with 2 generators and 1 relation
        sage: x, y = M.gens()
        sage: 2*x + 3*y == 0
        True
        
        sage: # Abelian group Z/2Z × Z/3Z
        sage: M = quotient_by_relations(2, ['2*e0', '3*e1'], base_ring=ZZ)
        sage: M.invariant_factors()
        [2, 3]
    """
    # Determine base ring
    if base_ring is None:
        if relations:
            # Try to infer from relations
            base_ring = relations[0].parent().base_ring()
        else:
            base_ring = ZZ
    
    # Create free module on generators
    if isinstance(generators, (int, Integer)):
        free = FreeModule(base_ring, generators)
    else:
        free = FreeModule_with_basis(base_ring, basis=generators)
    
    # Parse relations
    if not relations:
        return free
    
    # Convert symbolic relations to module elements
    parsed_relations = []
    for rel in relations:
        if isinstance(rel, str):
            # Parse symbolic expression
            # This would need proper expression parsing
            raise NotImplementedError("Symbolic relation parsing")
        else:
            parsed_relations.append(free(rel))
    
    # Create quotient
    submodule = free.submodule(parsed_relations)
    return free.quotient(submodule)


def finitely_presented_module(presentation_matrix, base_ring=None):
    """
    Module from presentation matrix.
    
    Given matrix A, constructs cokernel of R^m → R^n.
    
    INPUT:
    - presentation_matrix -- m × n matrix of relations
    - base_ring -- base ring (inferred from matrix if not given)
    
    OUTPUT:
    Finitely presented module R^n/im(A)
    
    EXAMPLES::
    
        sage: # Module presented by matrix
        sage: A = matrix(ZZ, [[2, 0], [0, 3]])
        sage: M = finitely_presented_module(A)
        sage: M
        Finitely presented module by 2×2 matrix
        sage: M.invariant_factors()
        [2, 3]  # Z/2Z × Z/3Z
    """
    if base_ring is None:
        base_ring = presentation_matrix.base_ring()
    
    n = presentation_matrix.ncols()
    m = presentation_matrix.nrows()
    
    # Target module R^n
    target = FreeModule(base_ring, n)
    
    # Relations are rows of matrix
    relations = [sum(presentation_matrix[i,j] * target.gen(j) 
                    for j in range(n))
                for i in range(m)]
    
    # Quotient by relations
    rel_submodule = target.submodule(relations)
    return target.quotient(rel_submodule)
```

```python
def graded_module(base_ring, grading_group=ZZ, degrees=None):
    """
    Construct graded module.
    
    M = ⊕_{g ∈ G} M_g where G is the grading group.
    
    INPUT:
    - base_ring -- base ring R
    - grading_group -- group G for grading (default: Z)
    - degrees -- dict of degree → module or generating data
    
    OUTPUT:
    G-graded R-module
    
    EXAMPLES::
    
        sage: # Z-graded module (polynomial ring structure)
        sage: M = graded_module(QQ, degrees={0: 1, 1: 2, 2: 1})
        sage: M
        Z-graded module over Rational Field
        sage: M.graded_component(1).rank()
        2
    """
    from sage.modules.graded_module import GradedModule
    
    if degrees is None:
        degrees = {}
    
    # Convert degree data to modules
    components = {}
    for deg, data in degrees.items():
        if isinstance(data, (int, Integer)):
            # Rank specified
            components[deg] = FreeModule(base_ring, data)
        else:
            # Assume it's already a module
            components[deg] = data
    
    return GradedModule(base_ring, grading_group, components)


def filtered_module(base_ring, filtration):
    """
    Construct filtered module.
    
    ... ⊆ F_{-1}M ⊆ F_0M ⊆ F_1M ⊆ ...
    
    INPUT:
    - base_ring -- base ring R
    - filtration -- dict of index → submodule
    
    OUTPUT:
    Filtered R-module
    
    EXAMPLES::
    
        sage: M = RModule(QQ, 3)
        sage: F = filtered_module(QQ, {
        ....:     0: M.submodule([M.gen(0)]),
        ....:     1: M.submodule([M.gen(0), M.gen(1)]),
        ....:     2: M
        ....: })
        sage: F
        Filtered module over Rational Field
        sage: F.graded_pieces()  # F_i/F_{i-1}
        [Rank 1 module, Rank 1 module, Rank 1 module]
    """
    from sage.modules.filtered_module import FilteredModule
    return FilteredModule(base_ring, filtration)


def chain_complex_module(differentials, base_ring=None):
    """
    Module with differential (chain complex in degree 0).
    
    Structure: M with d: M → M satisfying d² = 0.
    
    INPUT:
    - differentials -- endomorphism d or list of differentials
    - base_ring -- base ring
    
    OUTPUT:
    Differential module
    
    EXAMPLES::
    
        sage: M = RModule(QQ, 3)
        sage: # Differential that squares to zero
        sage: d = M.hom([M.gen(1), M.gen(2), 0])
        sage: C = chain_complex_module(d)
        sage: C
        Differential module over Rational Field
        sage: C.homology()
        Vector space of dimension 1 over Rational Field  # ker(d)/im(d)
    """
    # This would create a chain complex concentrated in degree 0
    from sage.homology.chain_complex import ChainComplex
    
    if hasattr(differentials, 'domain'):
        # Single differential
        module = differentials.domain()
        base_ring = module.base_ring()
        
        # Verify d² = 0
        d_squared = differentials * differentials
        if not d_squared.is_zero():
            raise ValueError("Differential must square to zero")
        
        # Create chain complex ... → 0 → M → M → 0 → ...
        data = {0: differentials}
        return ChainComplex(data, base_ring=base_ring)
    else:
        # Multiple differentials - create longer complex
        raise NotImplementedError("Multi-differential complexes")
```

```python
def is_valid_module_construction(module):
    """
    Validate that a constructed module satisfies basic axioms.
    
    Checks module axioms and category membership.
    
    INPUT:
    - module -- purported R-module
    
    OUTPUT:
    Boolean indicating validity
    
    EXAMPLES::
    
        sage: M = RModule(ZZ, 3)
        sage: is_valid_module_construction(M)
        True
    """
    try:
        # Check has base ring
        R = module.base_ring()
        
        # Check has module operations
        if hasattr(module, 'gens'):
            gens = module.gens()
            if gens:
                # Test module operations
                g = gens[0]
                zero = module.zero()
                
                # Addition
                _ = g + g
                _ = g + zero
                
                # Scalar multiplication
                _ = R.one() * g
                _ = R.zero() * g
        
        # Check category
        from sage.categories.modules import Modules
        return module in Modules(R)
        
    except:
        return False
```
