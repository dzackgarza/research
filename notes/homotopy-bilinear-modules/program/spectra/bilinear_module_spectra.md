class BilinearModuleSpectrum:
    r"""
    A spectrum object in the stable ∞-category Sp(BilR-Mod).
    
    Concretely, a bilinear module spectrum E consists of:
    - A sequence of bilinear modules {E_n}_{n≥0}
    - Structure maps σ_n: ΣE_n → E_{n+1} (or adjointly ε_n: E_n → ΩE_{n+1})
    - Higher coherence data making this an object of Sp(BilR-Mod)
    
    This is the bilinear analog of the notion of spectrum in stable homotopy theory.
    """
    
    def __init__(self, levels, structure_maps, coherence_data=None):
        """
        Initialize a bilinear module spectrum.
        
        INPUT:
        - levels: Dict {n: E_n} of bilinear modules
        - structure_maps: Dict {n: σ_n} where σ_n: ΣE_n → E_{n+1}
        - coherence_data: Higher homotopical data (for non-strict spectra)
        """
        self.levels = levels
        self.structure_maps = structure_maps
        self.coherence_data = coherence_data
    
    def level(self, n):
        r"""
        Return the n-th level E_n of the spectrum.
        
        EXAMPLES::

        - M: A bilinear module

            levels[n] = current if n == n_start else Suspension(current)
            if n > n_start:
                # Structure map is essentially identity after suspension
                maps[n-1] = StructureMap.identity_suspension(levels[n-1], levels[n])
            current = levels[n]
        
        return cls(levels, maps)
